В рамках лабораторной работы реализовано одностраничное приложение (SPA) на Vue 3 для работы с Django REST Framework бэкендом, использующим Djoser для аутентификации. Проект использует современный стек технологий для создания интерактивного пользовательского интерфейса.

Архитектура проекта

Проект построен на следующих технологиях:

  • Vue 3 (Composition API) - реактивный фреймворк для создания пользовательских интерфейсов
  • Vue Router - клиентская маршрутизация
  • Pinia - управление состоянием приложения
  • Axios - HTTP клиент для взаимодействия с API
  • Vuetify 3 - Material Design компоненты для UI
  • Vite - быстрый сборщик и dev-сервер

Реализация аутентификации на фронтенде

API модуль для аутентификации (src/api/auth.js)

Создан отдельный экземпляр Axios для публичных запросов (регистрация, вход), чтобы избежать автоматического добавления токена:

import axios from 'axios'

const BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:8000'

// Чистый экземпляр axios без interceptors
const cleanAxios = axios.create({
  baseURL: BASE_URL,
  headers: {
    'Content-Type': 'application/json',
  },
})

/**
 * Вход пользователя
 * @param {string} username - Имя пользователя
 * @param {string} password - Пароль
 * @returns {Promise} { auth_token: "..." }
 */
export const login = async (username, password) => {
  const response = await cleanAxios.post('/auth/token/login/', {
    username,
    password,
  })
  return response.data
}

/**
 * Регистрация нового пользователя
 * @param {string} username - Имя пользователя
 * @param {string} email - Email (опционально)
 * @param {string} password - Пароль
 * @returns {Promise} Данные созданного пользователя
 */
export const register = async (username, email, password) => {
  const payload = { username, password }
  if (email) {
    payload.email = email
  }
  const response = await cleanAxios.post('/auth/users/', payload)
  return response.data
}

/**
 * Получение информации о текущем пользователе
 * @param {string} token - Токен аутентификации
 * @returns {Promise} Данные пользователя
 */
export const getCurrentUser = async (token) => {
  const response = await cleanAxios.get('/auth/users/me/', {
    headers: {
      'Authorization': `Token ${token}`,
    },
  })
  return response.data
}

/**
 * Выход (аннулирование токена)
 * @param {string} token - Токен аутентификации
 */
export const logout = async (token) => {
  await cleanAxios.post('/auth/token/logout/', null, {
    headers: {
      'Authorization': `Token ${token}`,
    },
  })
}

Pinia Store для аутентификации (src/stores/authStore.js)

Используется Pinia с плагином pinia-plugin-persistedstate для сохранения состояния в localStorage:

import { defineStore } from 'pinia'
import * as authAPI from '@/api/auth'

export const useAuthStore = defineStore('auth', {
  state: () => ({
    token: null,
    user: null,
    isLoading: false,
  }),

  getters: {
    isAuthenticated: (state) => !!state.token && !!state.user,
  },

  actions: {
    async login(username, password) {
      this.isLoading = true
      try {
        const data = await authAPI.login(username, password)
        this.token = data.auth_token
        // Загружаем пользователя с явным токеном
        const user = await authAPI.getCurrentUser(this.token)
        this.user = user
        return { success: true }
      } catch (error) {
        const message =
          error.response?.data?.non_field_errors?.[0] ||
          error.response?.data?.detail ||
          'Ошибка входа'
        return { success: false, error: message }
      } finally {
        this.isLoading = false
      }
    },

    async register(username, email, password) {
      this.isLoading = true
      try {
        await authAPI.register(username, email, password)
        // После регистрации сразу логинимся
        return await this.login(username, password)
      } catch (error) {
        const errors = error.response?.data || {}
        let message = 'Ошибка регистрации'
        if (errors.username) {
          message = Array.isArray(errors.username) ? errors.username[0] : errors.username
        } else if (errors.password) {
          message = Array.isArray(errors.password) ? errors.password[0] : errors.password
        } else if (errors.email) {
          message = Array.isArray(errors.email) ? errors.email[0] : errors.email
        }
        return { success: false, error: message }
      } finally {
        this.isLoading = false
      }
    },

    async logout() {
      if (this.token) {
        try {
          await authAPI.logout(this.token)
        } catch (e) {
          // игнорируем ошибки logout
        }
      }
      this.token = null
      this.user = null
    },
  },

  persist: {
    enabled: true,
    strategies: [
      {
        key: 'auth',
        storage: localStorage,
        paths: ['token', 'user'],
      },
    ],
  },
})

Настройка Axios Interceptor (src/api/axios.js)

Основной экземпляр Axios автоматически добавляет токен к каждому запросу:

import axios from 'axios'
import { useAuthStore } from '@/stores/authStore'

const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || 'http://localhost:8000',
  headers: {
    'Content-Type': 'application/json',
  },
})

// Добавляем токен к каждому запросу
apiClient.interceptors.request.use((config) => {
  const authStore = useAuthStore()
  const publicEndpoints = ['/auth/users/', '/auth/token/login/']
  const isPublic = publicEndpoints.some((e) => config.url?.includes(e))

  if (!isPublic && authStore.token) {
    config.headers.Authorization = `Token ${authStore.token}`
  }
  return config
})

export default apiClient

Middleware для отключения CSRF

Для API endpoints отключена проверка CSRF:

# backend/middleware.py
class DisableCSRFForAPI:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        if request.path.startswith('/api/') or request.path.startswith('/auth/'):
            setattr(request, '_dont_enforce_csrf_checks', True)
        return self.get_response(request)

Route Guards

Реализованы guards для защиты маршрутов:

router.beforeEach(async (to, from, next) => {
  const authStore = useAuthStore()

  // Если маршрут требует аутентификации
  if (to.meta.requiresAuth) {
    if (!authStore.isAuthenticated) {
      // Пытаемся восстановить сессию из localStorage
      if (authStore.token) {
        try {
          await authStore.loadUser()
          if (authStore.isAuthenticated) {
            next()
            return
          }
        } catch (error) {
          authStore.logout()
        }
      }
      next({ name: 'login', query: { redirect: to.fullPath } })
      return
    }
    next()
    return
  }

  // Если маршрут доступен только для гостей
  if (to.meta.requiresGuest) {
    if (authStore.isAuthenticated) {
      next({ name: 'home' })
      return
    }
    next()
    return
  }

  next()
})

  • GET /api/documents/ - список всех документов
  • POST /api/documents/ - создание нового документа
  • GET /api/documents/<id>/ - детальная информация о документе
  • PATCH /api/documents/<id>/ - обновление документа
  • DELETE /api/documents/<id>/ - удаление документа

Особенности реализации

Автоматическое добавление токена

Axios interceptor автоматически добавляет токен к каждому запросу:

apiClient.interceptors.request.use((config) => {
  const authStore = useAuthStore()
  const publicEndpoints = ['/auth/users/', '/auth/token/login/']
  const isPublic = publicEndpoints.some((e) => config.url?.includes(e))

  if (!isPublic && authStore.token) {
    config.headers.Authorization = `Token ${authStore.token}`
  }
  return config
})

Сохранение состояния

Используется pinia-plugin-persistedstate для сохранения токена и данных пользователя в localStorage:

persist: {
  enabled: true,
  strategies: [
    {
      key: 'auth',
      storage: localStorage,
      paths: ['token', 'user'],
    },
  ],
}

Валидация форм

Все формы используют встроенную валидацию Vuetify:

const usernameRules = [
  (v) => !!v || 'Имя пользователя обязательно',
  (v) => (v && v.length >= 3) || 'Минимум 3 символа',
]

const passwordRules = [
  (v) => !!v || 'Пароль обязателен',
  (v) => (v && v.length >= 8) || 'Пароль должен содержать минимум 8 символов',
]

Обработка загрузки данных

Все компоненты показывают индикатор загрузки:

<v-card v-if="loading">
  <v-card-text>
    <v-progress-circular
      indeterminate
      color="primary"
      class="d-block mx-auto"
    ></v-progress-circular>
  </v-card-text>
</v-card>

Запуск проекта

Установка зависимостей

npm install

Настройка переменных окружения

Создайте файл .env в корне проекта:

VITE_API_BASE_URL=http://localhost:8000

Запуск dev-сервера

npm run dev

Приложение будет доступно по адресу http://localhost:5173

Сборка для продакшена

npm run build

Собранные файлы будут в папке dist/

Безопасность

  • Все API запросы (кроме регистрации и входа) требуют токен аутентификации
  • Токен автоматически добавляется к каждому запросу через Axios interceptor
  • Защита маршрутов через route guards
  • Валидация всех форм на клиенте
  • CORS настроен только для разрешённых доменов
  • CSRF отключен для API endpoints (используется Token Authentication)
